   202→   202→// ═══════════════════════════════════════════════════════════════════════════════
   203→   203→
   204→   204→function ImageSimulationTab() {
   205→   205→  const [srcGamut, setSrcGamut] = useState<string>('sRGB');
   206→   206→  const [srcTF, setSrcTF] = useState<TransferFunctionName>('sRGB');
   207→   207→  const [dstGamut, setDstGamut] = useState<string>('DCI_P3');
   208→   208→  const [dstTF, setDstTF] = useState<TransferFunctionName>('sRGB');
   209→   209→  const [imageSrc, setImageSrc] = useState<string | null>(null);
   210→   210→  const [simulatedSrc, setSimulatedSrc] = useState<string | null>(null);
   211→   211→  const [isDragging, setIsDragging] = useState(false);
   212→   212→  const [isSimulating, setIsSimulating] = useState(false);
   213→   213→  const [imageInfo, setImageInfo] = useState<{ width: number; height: number } | null>(null);
   214→   214→  const [simExportFormat, setSimExportFormat] = useState<ExportImageFormat>('png');
   215→   215→
   216→   216→  const fileInputRef = useRef<HTMLInputElement>(null);
   217→   217→  const imageRef = useRef<HTMLImageElement | null>(null);
   218→   218→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   219→   219→  const hiddenCanvasRef = useRef<HTMLCanvasElement | null>(null);
   220→   220→  const imageFileRef = useRef<File | null>(null); // keep File for raw loading
   221→   221→
   222→   222→  const handleFile = useCallback((file: File) => {
   223→   223→    // Allow image/* types and EXR/TIFF files (which may not have image/ MIME type)
   224→   224→    const isExrOrTiff = file.name.toLowerCase().endsWith('.exr') || file.name.toLowerCase().endsWith('.tiff') || file.name.toLowerCase().endsWith('.tif');
   225→   225→    if (!file.type.startsWith('image/') && !isExrOrTiff) return;
   226→   226→    imageFileRef.current = file;
   227→   227→    setSimulatedSrc(null);
   228→   228→    setImageInfo(null);
   229→   229→    // Load without ICC conversion for correct preview
   230→   230→    loadImageRaw(file).then((result) => {
   231→   231→      setImageSrc(result.dataUrl);
   232→   232→      setImageInfo({ width: result.width, height: result.height });
   233→   233→    }).catch(() => {
   234→   234→      // Fallback to blob URL
   235→   235→      setImageSrc(URL.createObjectURL(file));
   236→   236→    });
   237→   237→  }, []);
   238→   238→
   239→   239→  const handleDrop = useCallback(
   240→   240→    (e: React.DragEvent) => {
   241→   241→      e.preventDefault();
   242→   242→      setIsDragging(false);
   243→   243→      const file = e.dataTransfer.files[0];
   244→   244→      if (file) handleFile(file);
   245→   245→    },
   246→   246→    [handleFile]
   247→   247→  );
   248→   248→
   249→   249→  const handleDragOver = useCallback((e: React.DragEvent) => {
   250→   250→    e.preventDefault();
   251→   251→    setIsDragging(true);
   252→   252→  }, []);
   253→   253→
   254→   254→  const handleDragLeave = useCallback(() => setIsDragging(false), []);
   255→   255→
   256→   256→  const handleImageLoad = useCallback(() => {
   257→   257→    if (imageRef.current) {
   258→   258→      setImageInfo({
   259→   259→        width: imageRef.current.naturalWidth,
   260→   260→        height: imageRef.current.naturalHeight,
   261→   261→      });
   262→   262→    }
   263→   263→  }, []);
   264→   264→
   265→   265→  const simulateImage = useCallback(() => {
   266→   266→    if (!imageFileRef.current) return;
   267→   267→    setIsSimulating(true);
   268→   268→
   269→   269→    // Load image without ICC conversion for correct pixel processing
   270→   270→    loadImageRaw(imageFileRef.current).then((rawResult) => {
   271→   271→      try {
   272→   272→        const canvas = hiddenCanvasRef.current;
   273→   273→        if (!canvas) return;
   274→   274→        canvas.width = rawResult.width;
   275→   275→        canvas.height = rawResult.height;
   276→   276→        const ctx = canvas.getContext('2d');
   277→   277→        if (!ctx) return;
   278→   278→        ctx.drawImage(rawResult.canvas, 0, 0);
   279→   279→        const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
   280→   280→
   281→   281→        const lut = createColorSpaceLUT(33, srcGamut, srcTF, dstGamut, dstTF);
   282→   282→        const result = applyLUTToImageData(lut, imageData);
   283→   283→        ctx.putImageData(result, 0, 0);
   284→   284→
   285→   285→        setSimulatedSrc(canvas.toDataURL('image/png'));
   286→   286→      } catch {
   287→   287→        // silently handle errors
   288→   288→      } finally {
   289→   289→        setIsSimulating(false);
   290→   290→      }
   291→   291→    }).catch(() => {
   292→   292→      setIsSimulating(false);
   293→   293→    });
   294→   294→  }, [srcGamut, srcTF, dstGamut, dstTF]);
   295→   295→
   296→   296→  const handleDownloadSimulated = useCallback(async () => {
   297→   297→    if (!simulatedSrc) return;
   298→   298→    try {
   299→   299→      await exportDataUrlAsFormat(
   300→   300→        simulatedSrc,
   301→   301→        `simulated_${srcGamut}_to_${dstGamut}`,
   302→   302→        simExportFormat
   303→   303→      );
   304→   304→    } catch {
   305→   305→      // Fallback to simple PNG download
   306→   306→      const a = document.createElement('a');
   307→   307→      a.href = simulatedSrc;
   308→   308→      a.download = `simulated_${srcGamut}_to_${dstGamut}.png`;
   309→   309→      a.click();
   310→   310→    }
   311→   311→  }, [simulatedSrc, srcGamut, dstGamut, simExportFormat]);
   312→   312→
   313→   313→  return (
   314→   314→    <div className="space-y-4">
   315→   315→      {/* Controls */}
   316→   316→      <Card>
   317→   317→        <CardHeader className="pb-3">
   318→   318→          <CardTitle className="text-base">色彩空间转换仿真</CardTitle>
   319→   319→          <CardDescription>上传图片，选择源/目标色彩空间，预览转换效果</CardDescription>
   320→   320→        </CardHeader>
   321→   321→        <CardContent className="space-y-4">
   322→   322→          {/* Gamut / TF selectors */}
   323→   323→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   324→   324→            <div className="space-y-2">
   325→   325→              <Label className="text-sm font-medium">源色彩空间</Label>
   326→   326→              <div className="grid grid-cols-2 gap-2">
   327→   327→                <Select value={srcGamut} onValueChange={setSrcGamut}>
   328→   328→                  <SelectTrigger className="h-9">
   329→   329→                    <SelectValue placeholder="色域" />
   330→   330→                  </SelectTrigger>
   331→   331→                  <SelectContent>
   332→   332→                    {gamutNames.map((g) => (
   333→   333→                      <SelectItem key={g} value={g}>
   334→   334→                        {STANDARD_GAMUTS[g]?.name || g}
   335→   335→                      </SelectItem>
   336→   336→                    ))}
   337→   337→                  </SelectContent>
   338→   338→                </Select>
   339→   339→                <Select value={srcTF} onValueChange={(v) => setSrcTF(v as TransferFunctionName)}>
   340→   340→                  <SelectTrigger className="h-9">
   341→   341→                    <SelectValue placeholder="传输函数" />
   342→   342→                  </SelectTrigger>
   343→   343→                  <SelectContent>
   344→   344→                    {tfNames.map((tf) => (
   345→   345→                      <SelectItem key={tf} value={tf}>
   346→   346→                        {tf}
   347→   347→                      </SelectItem>
   348→   348→                    ))}
   349→   349→                  </SelectContent>
   350→   350→                </Select>
   351→   351→              </div>
   352→   352→            </div>
   353→   353→            <div className="space-y-2">
   354→   354→              <Label className="text-sm font-medium">目标色彩空间</Label>
   355→   355→              <div className="grid grid-cols-2 gap-2">
   356→   356→                <Select value={dstGamut} onValueChange={setDstGamut}>
   357→   357→                  <SelectTrigger className="h-9">
   358→   358→                    <SelectValue placeholder="色域" />
   359→   359→                  </SelectTrigger>
   360→   360→                  <SelectContent>
   361→   361→                    {gamutNames.map((g) => (
   362→   362→                      <SelectItem key={g} value={g}>
   363→   363→                        {STANDARD_GAMUTS[g]?.name || g}
   364→   364→                      </SelectItem>
   365→   365→                    ))}
   366→   366→                  </SelectContent>
   367→   367→                </Select>
   368→   368→                <Select value={dstTF} onValueChange={(v) => setDstTF(v as TransferFunctionName)}>
   369→   369→                  <SelectTrigger className="h-9">
   370→   370→                    <SelectValue placeholder="传输函数" />
   371→   371→                  </SelectTrigger>
   372→   372→                  <SelectContent>
   373→   373→                    {tfNames.map((tf) => (
   374→   374→                      <SelectItem key={tf} value={tf}>
   375→   375→                        {tf}
   376→   376→                      </SelectItem>
   377→   377→                    ))}
   378→   378→                  </SelectContent>
   379→   379→                </Select>
   380→   380→              </div>
   381→   381→            </div>
   382→   382→          </div>
   383→   383→
   384→   384→          {/* Pipeline info */}
   385→   385→          <div className="flex items-center gap-2 text-xs text-muted-foreground">
   386→   386→            <span className="font-medium text-foreground">{STANDARD_GAMUTS[srcGamut]?.name || srcGamut}</span>
   387→   387→            <span>+ {srcTF}</span>
   388→   388→            <span className="mx-1">→</span>
   389→   389→            <span className="font-medium text-foreground">{STANDARD_GAMUTS[dstGamut]?.name || dstGamut}</span>
   390→   390→            <span>+ {dstTF}</span>
   391→   391→          </div>
   392→   392→
   393→   393→          <Separator />
   394→   394→
   395→   395→          {/* Image upload */}
   396→   396→          <div
   397→   397→            onDrop={handleDrop}
   398→   398→            onDragOver={handleDragOver}
   399→   399→            onDragLeave={handleDragLeave}
   400→   400→            onClick={() => fileInputRef.current?.click()}
   401→   401→            className={`border-2 border-dashed rounded-lg p-8 text-center cursor-pointer transition-colors ${
   402→   402→              isDragging
   403→   403→                ? 'border-primary bg-primary/5'
   404→   404→                : 'border-muted-foreground/25 hover:border-primary/50'
   405→   405→            }`}
   406→   406→          >
   407→   407→            <input
   408→   408→              ref={fileInputRef}
   409→   409→              type="file"
   410→   410→              accept={IMAGE_ACCEPT_STRING}
   411→   411→              className="hidden"
   412→   412→              onChange={(e) => {
   413→   413→                const file = e.target.files?.[0];
   414→   414→                if (file) handleFile(file);
   415→   415→              }}
   416→   416→            />
   417→   417→            <Upload className="h-8 w-8 mx-auto mb-2 text-muted-foreground" />
   418→   418→            <p className="text-sm text-muted-foreground">
   419→   419→              拖拽图片到此处或点击上传
   420→   420→            </p>
   421→   421→            <p className="text-xs text-muted-foreground/60 mt-0.5">
   422→   422→              支持 PNG / JPEG / WebP / TIFF / EXR
   423→   423→            </p>
   424→   424→            {imageInfo && (
   425→   425→              <p className="text-xs text-muted-foreground mt-1">
   426→   426→                {imageInfo.width} × {imageInfo.height}px
   427→   427→              </p>
   428→   428→            )}
   429→   429→          </div>
   430→   430→
   431→   431→          {/* Simulate button */}
   432→   432→          <div className="flex gap-2">
   433→   433→            <Button
   434→   434→              onClick={simulateImage}
   435→   435→              disabled={!imageSrc || isSimulating}
   436→   436→              className="gap-1"
   437→   437→            >
   438→   438→              <Play className="h-3.5 w-3.5" />
   439→   439→              {isSimulating ? '处理中...' : '仿真'}
   440→   440→            </Button>
   441→   441→            {simulatedSrc && (
   442→   442→              <div className="flex items-center gap-2">
   443→   443→                <Select value={simExportFormat} onValueChange={(v) => setSimExportFormat(v as ExportImageFormat)}>
   444→   444→                  <SelectTrigger className="h-8 w-28 text-xs">
   445→   445→                    <SelectValue />
   446→   446→                  </SelectTrigger>
   447→   447→                  <SelectContent>
   448→   448→                    {EXPORT_FORMAT_OPTIONS.map((opt) => (
   449→   449→                      <SelectItem key={opt.value} value={opt.value} className="text-xs">
   450→   450→                        {opt.label}
   451→   451→                      </SelectItem>
   452→   452→                    ))}
   453→   453→                  </SelectContent>
   454→   454→                </Select>
   455→   455→                <Button variant="outline" onClick={handleDownloadSimulated} className="gap-1">
   456→   456→                  <Download className="h-3.5 w-3.5" />
   457→   457→                  下载结果
   458→   458→                </Button>
   459→   459→              </div>
   460→   460→            )}
   461→   461→          </div>
   462→   462→        </CardContent>
   463→   463→      </Card>
   464→   464→
   465→   465→      {/* Side-by-side display */}
   466→   466→      {imageSrc && (
   467→   467→        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   468→   468→          <Card>
   469→   469→            <CardHeader className="pb-2">
   470→   470→              <CardTitle className="text-sm flex items-center gap-2">
